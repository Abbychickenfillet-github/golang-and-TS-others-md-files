# Development Dockerfile with Air for hot reload
FROM golang:1.26-alpine

RUN apk add --no-cache git ca-certificates tzdata wget

WORKDIR /app

# Install Air for hot reload
RUN go install github.com/air-verse/air@latest

# Copy go mod files
COPY go.mod go.sum ./
RUN go mod download

COPY . .

EXPOSE 8080

CMD ["air", "-c", ".air.toml"]
apk 指令意思
apk add 套件安裝套件
apk del 套件移除套件
apk search 關鍵字搜尋套件
apk info -s 套件查看已安裝套件資訊（例如 tzdata-2026e）

小測驗：容器內 date 顯示 UTC，已經裝了 tzdata，為什麼時間還是 UTC？

tzdata 只是「提供時區資料」，不會改變容器預設時區。要讓容器用台北時間，必須再設定環境變數 TZ=Asia/Taipei。
