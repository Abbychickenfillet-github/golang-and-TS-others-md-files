② 證明給自己看：文字節點裡沒有 title

const el = document.querySelector('abbr');
el.textContent          // "HTML"          ← 只有這四個字
el.title                // "HyperText Markup Language"
el.getAttribute('title')// "HyperText Markup Language"
el.childNodes.length    // 1，只有一個文字節點
